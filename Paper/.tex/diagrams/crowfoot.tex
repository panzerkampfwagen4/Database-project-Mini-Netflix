% Helper macro for ER entity tables with a styled header and left-aligned attributes
% #1 = node name, #2 = x, #3 = y, #4 = header fill, #5 = border color, #6 = width, #7 = title, #8 = attribute rows
\newcommand{\ertable}[8]{%
  \node[
    rectangle,
    draw=#5,
    fill=white,
    line width=0.9pt,
    inner sep=0pt,
    font=\sffamily\scriptsize,
    align=left
  ] (#1) at (#2, #3) {%
    \begin{tabular}{@{}p{#6}@{}}
      \multicolumn{1}{@{}>{\centering\arraybackslash\columncolor{#4}}p{#6}@{}}{\rule{0pt}{11pt}\textbf{#7}\rule[-4pt]{0pt}{4pt}} \\[-0.5pt]
      \hline
      #8
    \end{tabular}%
  };%
}

% Fallback if colortbl is not loaded: draw header background via TikZ node
\renewcommand{\ertable}[8]{%
  \node[
    rectangle,
    draw=#5,
    fill=white,
    line width=0.9pt,
    inner sep=0pt,
    font=\sffamily\scriptsize,
    align=left
  ] (#1) at (#2, #3) {%
    \begin{tabular}{@{}p{#6}@{}}
      \centering\textbf{#7}\rule{0pt}{11pt}\rule[-4pt]{0pt}{4pt} \tabularnewline
      \hline
      #8
    \end{tabular}%
  };
  \begin{pgfonlayer}{background}
    \fill[#4] (#1.north west) rectangle ($(#1.north east)+(0,-16.5pt)$);
  \end{pgfonlayer}
}

\pgfdeclarelayer{background}
\pgfsetlayers{background,main}
